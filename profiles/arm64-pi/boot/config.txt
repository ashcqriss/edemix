# Edemint Pi firmware config.
# 64-bit kernel, Pi 4 / Pi 5 friendly, KMS DRM for Wayland.

arm_64bit=1
arm_boost=1
enable_uart=0

# Modern GPU stack - required for Hyprland on Pi 4/5.
dtoverlay=vc4-kms-v3d
disable_overscan=1
gpu_mem=128

# seed-boot-firmware writes kernel8.img for Pi 4 and kernel_2712.img for Pi 5.
# Do not override kernel= here: the firmware must select the board-specific
# default name. The initrd has one common, explicitly configured filename.
auto_initramfs=1
initramfs initrd.img followkernel

[pi5]
arm_boost=1
# Active Cooler / fan HAT automatic control.
dtparam=cooling_fan=on

[all]
